구글 드라이브(스프레드시트) 실시간 데이터 누적 연동 가이드

구글 드라이브 앱스 스크립트(Google Apps Script)를 활용해 독립적인 무료 응답 저장 데이터베이스를 즉시 생성하는 단계입니다.

1

구글 스프레드시트 개설

구글 스프레드시트로 이동하여 빈 문서를 생성하고 적절한 파일명(예: 서울대 생성형 AI 의존성 조사 응답)을 지정합니다.

2

구글 앱스 스크립트 도구 켜기

스프레드시트 상단 메뉴에서 [확장 프로그램] > [Apps Script]를 클릭해 편집기 인터페이스를 로딩합니다.

3

기본 코드 대체 (복사 및 붙여넣기)

기존의 자동 생성 함수 블록을 완전히 지우고, 아래 제공된 코드를 마우스로 클릭해 전 복사하여 붙여넣습니다.

/* Google Apps Script: 학술용 다단계 설문 응답 실시간 구글 시트 저장 및 자동 확장 컬럼 주입 */
function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var postData = JSON.parse(e.postData.contents);
    
    // 첫 행 헤더 조회 및 신규 문항 ID 발생 시 자동 열 컬럼 추가
    var headers = sheet.getLastColumn() > 0 ? sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0] : [];
    
    // 고정 필드 헤더 체크 및 정렬
    var requiredHeaders = ["Timestamp", "UID"];
    for (var i = 0; i < requiredHeaders.length; i++) {
      if (headers.indexOf(requiredHeaders[i]) === -1) {
        headers.push(requiredHeaders[i]);
        sheet.getRange(1, headers.length).setValue(requiredHeaders[i]);
      }
    }
    
    // 전달받은 데이터 키값 분석 후 필요한 헤더 자동 보강
    var payloadKeys = Object.keys(postData);
    for (var j = 0; j < payloadKeys.length; j++) {
      var key = payloadKeys[j];
      if (headers.indexOf(key) === -1) {
        headers.push(key);
        sheet.getRange(1, headers.length).setValue(key);
      }
    }
    
    // 시트 헤더값 갱신 재로드
    headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
    
    // 신규 추가 행 데이터 준비
    var rowData = new Array(headers.length);
    rowData[headers.indexOf("Timestamp")] = new Date();
    rowData[headers.indexOf("UID")] = postData.UID || "Anonymous";
    
    for (var key in postData) {
      if (key !== "UID") {
        var val = postData[key];
        // 오브젝트 형식은 JSON 텍스트화하여 저장
        if (typeof val === 'object') {
          val = JSON.stringify(val);
        }
        rowData[headers.indexOf(key)] = val;
      }
    }
    
    // 데이터 행 추가 실행
    sheet.appendRow(rowData);
    
    // CORS 보안 규격 준수 반환값 생성
    return ContentService.createTextOutput(JSON.stringify({
      "result": "success",
      "message": "데이터가 전송되어 저장되었습니다."
    })).setMimeType(ContentService.MimeType.JSON);
    
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({
      "result": "error",
      "message": error.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

웹 앱(Web App) 배포 등록 및 URL 복사

1. 우측 상단의 푸른색 [배포] > [새 배포] 버튼을 선택합니다.
2. 배포 유형 기어 아이콘을 클릭하여 [웹 앱]을 지정합니다.
3. 권한 항목을 아래 규칙에 맞추어 정확히 수정합니다:
   - 웹 앱을 실행할 사용자: '나(본인의 구글 계정)'
   - 액세스할 수 있는 사용자: '모든 사용자(Anyone)' (응답 트래픽 수용을 위해 반드시 설정 필요)
4. 배포를 진행한 뒤 발급되는 웹 앱 URL (Web App URL) 주소를 마우스 드래그로 복사하여 왼쪽 탭 안의 입력 상자에 등록합니다.

GitHub 레포지토리 구축

GitHub에 로그인 후 완전히 비어있는 새 공개 레포지토리(Public Repository)를 개설합니다.

2

index.html 파일 커밋

왼쪽 하단의 '설문 패키지 HTML 다운로드' 버튼을 클릭하여 소스코드가 주입된 완성형 파일(index.html)을 컴퓨터에 보관한 뒤, 이 파일을 생성한 GitHub 레포지토리에 업로드(커밋)합니다.

3

GitHub Pages 웹서비스 구동

레포지토리의 상단 메뉴 중 [Settings] > [Pages] 탭으로 이동한 뒤, Build and deployment 영역의 Source를 Deploy from a branch로 선언하고 Branch를 main (/root) 로 맞춘 뒤 [Save] 버튼을 클릭합니다.

4

도메인 배포 주소 확인
